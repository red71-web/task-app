    <!-- JavaScript 邏輯與 API 串接 -->
    <script>
        // 後端 Google Apps Script API 部署網址
        const API_URL = "https://script.google.com/macros/s/AKfycbwejSSLN8mJV89Gt2kduk4Crdx26MVZeVE8BeVhlBvkzwbOUIB8rcTuK8BhwHM60ib8Mw/exec";

        let localTasks = [];
        let localHistories = [];

        // 初始化載入
        window.addEventListener('DOMContentLoaded', () => {
            fetchDataFromSheets();
        });

        // 1. 從 Google Sheets 抓取資料 (GET)
        async function fetchDataFromSheets() {
            updateSyncStatus(true, "連線資料庫中...");
            try {
                const response = await fetch(API_URL);
                const result = await response.json();
                
                localTasks = result.工作項目 || [];
                localHistories = result.處置履歷 || [];
                
                renderTasks();
                updateSyncStatus(false, "資料同步完成");
            } catch (error) {
                console.error("讀取失敗:", error);
                updateSyncStatus(false, "同步失敗，請重試", true);
            }
        }

        // 2. 渲染 UI 看板
        function renderTasks() {
            const container = document.getElementById('taskList');
            document.getElementById('taskCount').innerText = `共 ${localTasks.length} 項工作`;

            if (localTasks.length === 0) {
                container.innerHTML = `<p class="text-center text-slate-400 py-8 text-sm">目前試算表中尚無資料。</p>`;
                return;
            }

            container.innerHTML = localTasks.map(task => {
                const histories = localHistories.filter(h => h.任務編號 === task.任務編號);
                const priorityColor = task.優先級 === 'P0' ? 'bg-red-100 text-red-700 border-red-200' : 'bg-blue-100 text-blue-700 border-blue-200';
                
                return `
                    <div class="border border-slate-200 rounded-lg p-4 hover:shadow-md transition bg-white space-y-3">
                        <div class="flex justify-between items-start">
                            <div class="flex items-center space-x-2">
                                <span class="text-xs px-2 py-0.5 rounded border font-semibold ${priorityColor}">${task.優先級}</span>
                                <h3 class="font-bold text-slate-800 text-base">${escapeHtml(task.工作標題)}</h3>
                                ${task.快記膠囊 === '是' ? '<span class="text-xs bg-amber-100 text-amber-800 px-1.5 py-0.5 rounded">膠囊</span>' : ''}
                            </div>
                            <span class="text-xs font-semibold px-2 py-1 bg-slate-100 text-slate-600 rounded">${task.狀態}</span>
                        </div>
                        
                        <div class="text-xs text-slate-500 grid grid-cols-2 gap-2 bg-slate-50 p-2.5 rounded">
                            <div><i class="fa-solid fa-user mr-1 text-slate-400"></i>責任人: <span class="font-medium text-slate-700">${escapeHtml(task.責任人)}</span></div>
                            <div><i class="fa-solid fa-clock mr-1 text-slate-400"></i>時段: <span class="font-medium text-slate-700">${task.時段}</span></div>
                            <div><i class="fa-solid fa-diagram-project mr-1 text-slate-400"></i>工程階段: <span class="font-medium text-slate-700">${escapeHtml(task.工程階段 || '無')}</span></div>
                            <div><i class="fa-solid fa-calendar mr-1 text-slate-400"></i>建立日期: <span class="font-medium text-slate-700">${task.建立日期}</span></div>
                        </div>

                        <!-- 履歷記錄清單 -->
                        ${histories.length > 0 ? `
                            <div class="border-t border-slate-100 pt-2 space-y-1.5">
                                <span class="text-xs font-semibold text-slate-500">最新處置履歷：</span>
                                ${histories.map(h => `
                                    <div class="text-xs bg-amber-50/60 border border-amber-100 rounded p-2 text-slate-700">
                                        <div class="text-slate-400 text-[10px]">${h.記錄時間}</div>
                                        <div>${escapeHtml(h.處置說明)}</div>
                                        ${h.照片連結 ? `<a href="${h.照片連結}" target="_blank" class="text-indigo-600 underline text-[11px] block mt-1"><i class="fa-solid fa-image mr-1"></i>查看照片</a>` : ''}
                                    </div>
                                `).join('')}
                            </div>
                        ` : ''}

                        <div class="flex justify-end pt-1">
                            <button onclick="openHistoryModal('${task.任務編號}')" class="text-xs bg-slate-100 hover:bg-indigo-50 text-indigo-600 px-3 py-1.5 rounded font-medium transition flex items-center">
                                <i class="fa-solid fa-pen-to-square mr-1"></i> 新增處置履歷
                            </button>
                        </div>
                    </div>
                `;
            }).join('');
        }

        // 3. 處理表單提交 (新增工作)
        async function handleTaskSubmit(event) {
            event.preventDefault();
            
            const taskData = {
                任務編號: "T_" + new Date().getTime(),
                責任人: document.getElementById('owner').value,
                工作標題: document.getElementById('title').value,
                優先級: document.getElementById('priority').value,
                工程階段: document.getElementById('engStatus').value,
                時段: document.getElementById('slot').value,
                快記膠囊: document.getElementById('isCapsule').value,
                建立日期: new Date().toLocaleDateString('zh-TW'),
                預計完成日: "",
                預計時間: ""
            };

            updateSyncStatus(true, "正在新增工作至 Google Sheets...");
            
            try {
                await fetch(API_URL, {
                    method: "POST",
                    body: JSON.stringify({ 動作: "新增工作", 資料: taskData })
                });

                document.getElementById('taskForm').reset();
                await fetchDataFromSheets();
            } catch (error) {
                console.error("新增失敗:", error);
                updateSyncStatus(false, "新增失敗", true);
            }
        }

        // 4. Modal 控制與處置履歷寫入
        function openHistoryModal(taskId) {
            document.getElementById('modalTaskId').value = taskId;
            document.getElementById('modalHistoryNote').value = '';
            document.getElementById('modalPhotoUrl').value = '';
            document.getElementById('historyModal').classList.remove('hidden');
        }

        function closeModal() {
            document.getElementById('historyModal').classList.add('hidden');
        }

        async function handleHistorySubmit() {
            const taskId = document.getElementById('modalTaskId').value;
            const note = document.getElementById('modalHistoryNote').value;
            const photoUrl = document.getElementById('modalPhotoUrl').value;

            if (!note.trim()) {
                alert("請輸入處置說明");
                return;
            }

            const historyData = {
                任務編號: taskId,
                記錄時間: new Date().toLocaleString('zh-TW'),
                處置說明: note,
                照片連結: photoUrl
            };

            closeModal();
            updateSyncStatus(true, "正在儲存處置履歷...");

            try {
                await fetch(API_URL, {
                    method: "POST",
                    body: JSON.stringify({ 動作: "新增處置履歷", 資料: historyData })
                });

                await fetchDataFromSheets();
            } catch (error) {
                console.error("儲存履歷失敗:", error);
                updateSyncStatus(false, "儲存履歷失敗", true);
            }
        }

        // 狀態更新輔助函式
        function updateSyncStatus(isSyncing, text, isError = false) {
            const statusEl = document.getElementById('syncStatus');
            if (isSyncing) {
                statusEl.innerHTML = `<i class="fa-solid fa-circle-notch fa-spin mr-1 text-amber-400"></i> ${text}`;
            } else if (isError) {
                statusEl.innerHTML = `<i class="fa-solid fa-circle-exclamation mr-1 text-red-400"></i> ${text}`;
            } else {
                statusEl.innerHTML = `<i class="fa-solid fa-cloud-check mr-1 text-emerald-400"></i> ${text}`;
            }
        }

        function escapeHtml(str) {
            return (str || '').replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");
        }
    </script>
